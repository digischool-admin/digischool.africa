## 🗄️ BACKEND SUPABASE (supabase-client.js)

### Architecture Proposée

#### Tables
```sql
-- Leads / Prospects
CREATE TABLE leads (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  email TEXT UNIQUE NOT NULL,
  first_name TEXT,
  last_name TEXT,
  phone TEXT,
  source TEXT, -- 'b2c-assessment', 'checkout', 'b2b-form'
  created_at TIMESTAMP DEFAULT now(),
  metadata JSONB
);

-- Demandes de Paiement
CREATE TABLE payment_requests (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  lead_id UUID REFERENCES leads(id),
  selected_modules JSONB NOT NULL, -- [{id, name, price}, ...]
  total_price INTEGER NOT NULL,
  status TEXT DEFAULT 'pending', -- pending, verified, rejected
  created_at TIMESTAMP DEFAULT now()
);

-- Preuves de Paiement
CREATE TABLE payment_proofs (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  payment_request_id UUID REFERENCES payment_requests(id),
  payment_type TEXT NOT NULL, -- bank, mobile, western, other
  transaction_ref TEXT NOT NULL,
  proof_file_url TEXT, -- Supabase Storage URL
  comment TEXT,
  submitted_at TIMESTAMP DEFAULT now(),
  verified_at TIMESTAMP,
  verified_by UUID -- admin user
);

-- Droits d'Accès (Entitlements)
CREATE TABLE entitlements (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  lead_id UUID REFERENCES leads(id),
  module_id TEXT NOT NULL, -- 'gestion-projet-pmp', etc.
  granted_at TIMESTAMP DEFAULT now(),
  expires_at TIMESTAMP, -- NULL si illimité
  payment_request_id UUID REFERENCES payment_requests(id)
);

-- Audit Log
CREATE TABLE audit_log (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  entity_type TEXT NOT NULL, -- 'payment', 'entitlement', 'admin_action'
  entity_id UUID,
  action TEXT NOT NULL, -- 'created', 'updated', 'verified', 'granted'
  user_id UUID, -- admin ou système
  metadata JSONB,
  timestamp TIMESTAMP DEFAULT now()
);
```

#### RLS Policies (Row-Level Security)
```sql
-- Leads: utilisateurs voient seulement leurs propres données
ALTER TABLE leads ENABLE ROW LEVEL SECURITY;
CREATE POLICY "Users view own leads" ON leads
  FOR SELECT USING (auth.email() = email);

-- Payment Requests: idem
ALTER TABLE payment_requests ENABLE ROW LEVEL SECURITY;
CREATE POLICY "Users view own payment requests" ON payment_requests
  FOR SELECT USING (
    lead_id IN (SELECT id FROM leads WHERE email = auth.email())
  );

-- Entitlements: utilisateurs voient leurs droits
ALTER TABLE entitlements ENABLE ROW LEVEL SECURITY;
CREATE POLICY "Users view own entitlements" ON entitlements
  FOR SELECT USING (
    lead_id IN (SELECT id FROM leads WHERE email = auth.email())
  );

-- Admin: accès total (role admin)
CREATE POLICY "Admins full access" ON payment_proofs
  FOR ALL USING (auth.role() = 'admin');
```

#### Edge Functions (Supabase)
1. **`submit-payment-proof`**  
   → Stocke preuve dans `payment_proofs`  
   → Upload fichier vers Storage bucket `proofs/`  
   → Envoie notification email admin

2. **`validate-payment`** (Admin uniquement)  
   → Vérifie payment_proof  
   → Crée entitlements pour modules  
   → Envoie email confirmation utilisateur

3. **`check-access`**  
   → Vérifie si utilisateur a entitlement pour module  
   → Utilisé par course-player pour débloquer contenu
