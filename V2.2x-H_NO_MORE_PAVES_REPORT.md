## 🎓 COMPONENT USAGE GUIDE

### For Developers: Adding New Pages

```javascript
// 1. Include scripts in HTML
<script src="/assets/premium-ux-components-v2.2.x-h.js"></script>
<script src="/assets/apply-premium-ux-refactor.js"></script>

// 2. Use components in your code
const UX = window.DigiSchoolPremiumUX;

// Create a feature card
const card = UX.createFeatureCard({
  icon: '🎯',
  title: 'Titre',
  description: 'Description du feature',
  cta: { text: 'En savoir plus', href: '/link', style: 'primary' },
  badge: 'NOUVEAU'
});
document.getElementById('container').appendChild(card);

// Create an accordion
const accordion = UX.createAccordion({
  title: 'Section titre',
  content: '<p>Contenu détaillé</p>',
  defaultOpen: false
});

// Create tabs
const tabs = UX.createTabs({
  tabs: [
    { id: 'tab1', label: 'Option 1', content: '<p>Content 1</p>' },
    { id: 'tab2', label: 'Option 2', content: '<p>Content 2</p>' }
  ]
});

// Create CTA banner
const cta = UX.createCTABanner({
  title: 'Prêt à commencer ?',
  description: 'Transformez votre carrière',
  primaryCTA: { text: 'Commencer', href: '/start' }
});
```
